```typescript
function getLastCommandTime(state: any): number | null {
  const getLastTime = (commands: any[]): number | null => {
    const lastCommand = commands[commands.length - 1];
    return lastCommand ? lastCommand.unixMillisec : null;
  };

  const lastCommandTime = getLastTime(state.undoCommands);
  const lastSongCommandTime = getLastTime(state.undoSongCommands);

  return lastCommandTime && lastSongCommandTime

```
